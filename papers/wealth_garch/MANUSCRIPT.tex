\section{Introduction}

The question is how an asymmetric penalty on a capital shortfall shapes a
bank's recapitalisation policy and the volatility of its capital buffer.
The model treats recapitalisation as impulse control under a fixed issuance
cost and a regulatory cap on risky exposure.

\section{Proofs}
\label{app:proofs}

\paragraph{Notation.} In discrete time $V(z)$ is the conditional variance of
the shock at log buffer ratio $z$, with induced variance asymmetry
$\lambda_V>0$, saturation scale $\theta>0$ and level $V_0>0$. In continuous
time $x\in(1,\infty)$ is the ratio of assets to liabilities, $q>0$ the ratio
requirement, and $z(x)=\log\big((x-1)/q\big)$. The risky exposure is $\pi(x)$,
and the regulatory cap gives the capped exposure
$u(x)=\min\{(x-1)/a_1,(x-a_2)/a_3\}$ with crossing point
$\bar x=(a_3-a_1a_2)/(a_3-a_1)$. The diffusion of $x$ is
$\sigma^2(x,\pi)=\pi^2\sigma^2x^2+2\pi c\sigma\sigma_Lx(1-x)+\sigma_L^2(1-x)^2$,
with $c$ the correlation between the risky return and the liability shock.
The $z$-volatility is $\zeta(x)=z'(x)^2\,\sigma^2\big(x,\pi(x)\big)$, the
exposure ratio is $\omega(x)=\pi(x)\,x/(x-1)$, and $\kappa_i^2=\sigma^2/a_i^2-2c\sigma\sigma_L/a_i+\sigma_L^2$ for $i\in\{1,3\}$.

\begin{mproof}{M1}{RateBased.Vresp_ratio_eq, RateBased.tendsto_tanh_atTop, RateBased.saturated_ratio_tendsto, RateBased.saturated_ratio_lambda4}
For every $z$,
\[ \frac{V(-z)}{V(z)}=\exp\big(4\log\lambda_V\tanh(z/\theta)\big). \]
\[ \tanh y=1-\frac{2}{e^{2y}+1}\to1\quad(y\to\infty),\qquad z/\theta\to\infty\quad(z\to\infty,\ \theta>0). \]
\[ \exp\big(4\log\lambda_V\tanh(z/\theta)\big)\to\exp(4\log\lambda_V)=\lambda_V^4 . \]
$\theta$ enters only through $z/\theta\to\infty$, so the limit is free of $\theta$.
\end{mproof}

\begin{mproof}{M2}{CapGeometry.hasDerivAt_zmap, CapGeometry.zvol2_eq, CapGeometry.xbar_sub_one, CapGeometry.one_lt_xbar, CapGeometry.solvency_binds, CapGeometry.liquidity_binds, CapGeometry.zvol2_solvency}
\[ \frac{x-1}{a_1}\le\frac{x-a_2}{a_3}\iff(a_3-a_1)\,x\le a_3-a_1a_2\iff x\le\bar x,\qquad \bar x-1=\frac{a_1(1-a_2)}{a_3-a_1}>0 . \]
\[ z'(x)=\frac{1}{x-1},\qquad \zeta(x)=\omega(x)^2\sigma^2-2\,\omega(x)\,c\,\sigma\sigma_L+\sigma_L^2 . \]
\[ 1<x\le\bar x,\ \pi(x)\,x=u(x)\ \Rightarrow\ \omega(x)=\frac{1}{a_1}\ \Rightarrow\ \zeta(x)=\kappa_1^2. \]
\end{mproof}

\begin{mproof}{M3}{CapGeometry.deficit_limit, CapGeometry.surplus_limit, CapGeometry.variance_ratio_is_cap_geometry, CapGeometry.control_surplus_limit_needs_a1_lt_a3, CapGeometry.control_constancy_needs_binding}
Near $1^+$, $\bar x>1$ by M2, so $1<x<\bar x$ eventually and $\zeta(x)=\kappa_1^2$ there.
For $x\ge\bar x$,
\[ \omega(x)=\frac{x-a_2}{a_3(x-1)}=\frac{1}{a_3}+\frac{1-a_2}{a_3}\,(x-1)^{-1}\to\frac{1}{a_3}, \]
\[ \zeta(x)=\omega(x)^2\sigma^2-2\,\omega(x)\,c\,\sigma\sigma_L+\sigma_L^2\to\kappa_3^2 . \]
Controls, at $\sigma=1$, $\sigma_L=c=0$, $q=1$.
\[ a_1=\tfrac12,\ a_2=0,\ a_3=\tfrac14\ \Rightarrow\ u(x)=2(x-1),\ \zeta\equiv4\ne16=\kappa_3^2. \]
\[ \pi\equiv0\ \Rightarrow\ \zeta(2)=0\ne4=\kappa_1^2. \]
\end{mproof}

\section{Primitives and scope}
\label{app:primitives}

\paragraph{The regulatory cap.} The capped exposure is
$u(x)=\min\{(x-1)/a_1,(x-a_2)/a_3\}$ with $0<a_1<a_3$ and $a_2<1$, taken from
\path{PROOFS_v2.tex}. Under these restrictions the solvency branch binds near
the distress boundary and the liquidity branch deep in surplus (M2). With
$a_1\ge a_3$ the surplus limit of M3 fails, by its control. Rows M2, M3 and C1
depend on it.

\paragraph{The variance ratio in continuous time.} \path{PROOFS_v2.tex}
reads the discrete-time $\lambda_V^4$ of M1 in the continuous-time model as
$\kappa_1^2/\kappa_3^2$, identifying the conditional variance $V(z)$ deep in
deficit and deep in surplus with the two saturation limits of $\zeta$. This
is a definition, not a result, and no theorem connects the discrete-time
recursion to the continuous-time diffusion. C1 depends on it.
